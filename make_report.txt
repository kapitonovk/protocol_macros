Attribute VB_Name = "make_report"
Option Explicit

' ============================================================
' ОРКЕСТРАТОР СБОРКИ ПРОТОКОЛА
'
' Структура результата:
'   0. титульник            (Sources 0.1)
'   1. body x N             (Sources 0.2)  — по одному на каждый data-источник
'   2. лист согласования    (Sources 0.3)
'   3. appendix x N         (Sources 0.4)  — по одному на каждый data-источник
'
' Модуль make_appx НЕ изменяется.
'
' ---- ПРАВКИ 03.08.2026 (потеря оформления при мердже) ----
' Причина: Range.InsertFile не импортирует стили. Вставляемый текст
' сохраняет ИМЕНА стилей, но начинает подчиняться ОПРЕДЕЛЕНИЯМ стилей
' master-документа (он создаётся из титульника). Отсюда «плавающий» кегль
' в основном тексте, подзаголовках «Приложение» и таблицах, причём
' неоднородно: там, где в шаблоне было прямое форматирование поверх стиля,
' оно уцелело, там, где формат шёл от стиля — переопределился.
' Плюс InsertFile не переносит параметры страницы последней секции
' источника, поэтому альбомные таблицы приложения попадали в портретную
' секцию и «ехали».
'
' Решение:
'   1) вставка через буфер с сохранением исходного форматирования
'      (PasteAndFormat wdFormatOriginalFormatting) — Word превращает
'      конфликтующие стили в прямое форматирование, оформление не плывёт;
'   2) для каждой вставляемой части создаётся новая секция, в которую
'      копируется PageSetup первой секции источника, а после вставки
'      PageSetup последней секции источника копируется в последнюю секцию
'      master (её параметры хранятся в финальном ¶ и через буфер не едут);
'   3) колонтитулы новой секции отвязываются от предыдущей.
' ============================================================

' --- Поведенческие флаги -------------------------------------------------
Private Const FIX_HARDCODED_SECTION_NUMBERS As Boolean = False
Private Const DROP_DDU_BLOCK_FOR_NON_DDU As Boolean = True
Private Const SHOW_RESULT As Boolean = True

' Мердж через буфер с сохранением исходного оформления.
' False — старое поведение (Range.InsertFile), оставлено на случай отката.
Private Const MERGE_KEEP_SOURCE_FORMATTING As Boolean = True

' ---- ПРАВКА 2 от 03.08.2026 -------------------------------------------
' Master-документ создаётся НЕ из титульника, а из шаблона-донора стилей.
' У title.docx набор стилей другой родословной: одноимённые стили имеют
' другие styleId (Normal = "Normal" против "a1", Table Grid = "TableGrid"
' против "a7"), а один и тот же id значит разные стили (id "13" в титуле —
' "Сетка таблицы13" 14 пт, в body/appendix — основной текст 11 пт).
' body, approval и appendix при этом идентичны друг другу по стилям.
' Поэтому донором делаем body (0.2), а титульник вставляем как обычную
' часть — он на 100% свёрстан прямым форматированием и переносится нормально.
Private Const MASTER_STYLE_SOURCE_NUMBER As String = "0.2"

' Word по умолчанию для случая "вставка между документами при конфликте
' определений стилей" применяет Use Destination Styles — и переопределяет
' даже PasteAndFormat. Принудительно ставим Keep Source Formatting на время
' сборки и возвращаем прежние значения после.
Private Const FORCE_KEEP_SOURCE_PASTE_OPTIONS As Boolean = True

' Приводить стили каждой части к донорским перед сохранением.
' Крайняя мера: убирает конфликты полностью, но перебивает собственные
' определения стилей титульника.
Private Const COPY_STYLES_FROM_DONOR As Boolean = False

Private Const wdKeepSourceFormatting As Long = 0

' Собирать при видимом Word. Невидимый экземпляр откладывает раскладку,
' и таблицы с автоподбором по окну пересчитываются по неверной ширине.
' Именно поэтому отдельно собранный appendix при Visible = True выглядит нормально.
Private Const BUILD_WITH_WORD_VISIBLE As Boolean = True

' Переносить содержимое колонтитулов частей (если они непустые)
Private Const COPY_HEADERS_FOOTERS As Boolean = True

' --- Константы Word (позднее связывание) --------------------------------
Private Const wdCollapseStart As Long = 1
Private Const wdCollapseEnd As Long = 0
Private Const wdSectionBreakNextPage As Long = 2
Private Const wdFormatDocumentDefault As Long = 16
Private Const wdFormatOriginalFormatting As Long = 16
Private Const wdStory As Long = 6


' ============================================================
' ЕДИНСТВЕННАЯ ТОЧКА ВХОДА
' ============================================================
Public Sub BuildProtocol()

    Dim wbMain As Workbook
    Dim wsSources As Worksheet

    Dim srcNums() As String
    Dim srcRows() As Long
    Dim srcPaths() As String
    Dim srcCount As Long

    Dim titlePath As String
    Dim bodyPath As String
    Dim approvalPath As String
    Dim appendixPath As String

    Dim wdApp As Object
    Dim wdMaster As Object
    Dim wdPart As Object
    Dim wbData As Workbook

    Dim tmpDir As String
    Dim bodyFiles() As String
    Dim appxFiles() As String
    Dim approvalFile As String
    Dim titleFile As String
    Dim stylePath As String

    Dim optBetween As Long
    Dim optStyled As Long
    Dim optsSaved As Boolean

    Dim i As Long
    Dim outPath As String
    Dim prevScreen As Boolean
    Dim prevAlerts As Boolean

    Set wbMain = ThisWorkbook
    Set wsSources = wbMain.Worksheets("Sources")

    ' ---- 1. Пути к служебным шаблонам ----
    titlePath = GetSourcePathByNumber(wbMain, "0.1")
    bodyPath = GetSourcePathByNumber(wbMain, "0.2")
    approvalPath = GetSourcePathByNumber(wbMain, "0.3")
    appendixPath = GetSourcePathByNumber(wbMain, "0.4")

    If Not FileExists(titlePath) Then
        MsgBox "Не найден шаблон титульника (Sources 0.1): " & titlePath, vbExclamation
        Exit Sub
    End If
    If Not FileExists(bodyPath) Then
        MsgBox "Не найден шаблон body (Sources 0.2): " & bodyPath, vbExclamation
        Exit Sub
    End If
    If Not FileExists(approvalPath) Then
        MsgBox "Не найден шаблон листа согласования (Sources 0.3): " & approvalPath, vbExclamation
        Exit Sub
    End If
    If Not FileExists(appendixPath) Then
        MsgBox "Не найден шаблон приложения (Sources 0.4): " & appendixPath, vbExclamation
        Exit Sub
    End If

    ' ---- 2. Собираем список data-источников ----
    srcCount = CollectDataSources(wbMain, srcNums, srcRows, srcPaths)

    If srcCount = 0 Then
        MsgBox "В Sources не найдено ни одного существующего data-файла " & _
               "(строки с номерами 1, 2, 3 ...).", vbExclamation
        Exit Sub
    End If

    ReDim bodyFiles(1 To srcCount)
    ReDim appxFiles(1 To srcCount)

    tmpDir = MakeTempDir()
    If tmpDir = "" Then
        MsgBox "Не удалось создать временную папку для сборки.", vbExclamation
        Exit Sub
    End If

    prevScreen = Application.ScreenUpdating
    prevAlerts = Application.DisplayAlerts
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False

    On Error GoTo Fail

    ' ---- 3. Word, настройки вставки и пустой master из донора стилей ----
    Set wdApp = CreateObject("Word.Application")
    wdApp.Visible = BUILD_WITH_WORD_VISIBLE
    wdApp.DisplayAlerts = 0
    On Error Resume Next
    wdApp.ScreenUpdating = False
    On Error GoTo Fail

    If FORCE_KEEP_SOURCE_PASTE_OPTIONS Then
        On Error Resume Next
        Err.Clear
        optBetween = wdApp.Options.PasteFormatBetweenDocuments
        optStyled = wdApp.Options.PasteFormatBetweenStyledDocuments
        wdApp.Options.PasteFormatBetweenDocuments = wdKeepSourceFormatting
        wdApp.Options.PasteFormatBetweenStyledDocuments = wdKeepSourceFormatting
        optsSaved = (Err.Number = 0)
        On Error GoTo Fail
    End If

    stylePath = GetSourcePathByNumber(wbMain, MASTER_STYLE_SOURCE_NUMBER)
    If Not FileExists(stylePath) Then stylePath = bodyPath

    ' master = документ с донорскими стилями и пустым содержимым
    Set wdMaster = wdApp.Documents.Add(Template:=stylePath)
    wdMaster.Content.Delete

    ' ---- 4. По каждому источнику: body + appendix во временные файлы ----
    For i = 1 To srcCount

        Set wbData = Workbooks.Open(Filename:=srcPaths(i), ReadOnly:=True, UpdateLinks:=False)

        ' ---- body ----
        Set wdPart = wdApp.Documents.Add(Template:=bodyPath)
        Call BuildBodyDoc(wbMain, wbData, wdPart, srcNums(i), srcRows(i))
        Call ApplyConfigScalars(wbMain, wdPart)
        Call FinalizeSectionNumbers(wdPart, srcNums(i))
        Call TrimTrailingEmptyParagraphs(wdPart)
        If COPY_STYLES_FROM_DONOR Then wdPart.CopyStylesFromTemplate stylePath

        bodyFiles(i) = tmpDir & "part_body_" & Format(i, "00") & ".docx"
        wdPart.SaveAs2 Filename:=bodyFiles(i), FileFormat:=wdFormatDocumentDefault
        wdPart.Close SaveChanges:=False
        Set wdPart = Nothing

        ' ---- appendix ----
        Set wdPart = wdApp.Documents.Add(Template:=appendixPath)
        Call BuildSection_1_1_APPX(wbMain, wbData, wdPart, srcRows(i))

        If DROP_DDU_BLOCK_FOR_NON_DDU Then
            If UCase(Trim(GetSourceValueByHeader(wbMain, srcRows(i), "Тип"))) <> "ДДУ" Then
                Call DropDduInstalmentBlock(wdPart)
            End If
        End If

        Call ReplacePlaceholderInWord(wdPart, "{{КВАРТИРЫ_СЧЕТ}}", _
                                      CStr(CountVectorRows(wbMain, wbData, "APPX_112")))
        Call ApplyConfigScalars(wbMain, wdPart)
        Call FinalizeSectionNumbers(wdPart, srcNums(i))
        Call TrimTrailingEmptyParagraphs(wdPart)
        If COPY_STYLES_FROM_DONOR Then wdPart.CopyStylesFromTemplate stylePath

        appxFiles(i) = tmpDir & "part_appx_" & Format(i, "00") & ".docx"
        wdPart.SaveAs2 Filename:=appxFiles(i), FileFormat:=wdFormatDocumentDefault
        wdPart.Close SaveChanges:=False
        Set wdPart = Nothing

        wbData.Close SaveChanges:=False
        Set wbData = Nothing
    Next i

    ' ---- 4a. Титульник как обычная часть ----
    Set wdPart = wdApp.Documents.Add(Template:=titlePath)
    Call ApplyConfigScalars(wbMain, wdPart)
    Call TrimTrailingEmptyParagraphs(wdPart)
    If COPY_STYLES_FROM_DONOR Then wdPart.CopyStylesFromTemplate stylePath

    titleFile = tmpDir & "part_title.docx"
    wdPart.SaveAs2 Filename:=titleFile, FileFormat:=wdFormatDocumentDefault
    wdPart.Close SaveChanges:=False
    Set wdPart = Nothing

    ' ---- 5. Лист согласования во временный файл ----
    Set wdPart = wdApp.Documents.Add(Template:=approvalPath)
    Call ApplyConfigScalars(wbMain, wdPart)
    Call TrimTrailingEmptyParagraphs(wdPart)
    If COPY_STYLES_FROM_DONOR Then wdPart.CopyStylesFromTemplate stylePath

    approvalFile = tmpDir & "part_approval.docx"
    wdPart.SaveAs2 Filename:=approvalFile, FileFormat:=wdFormatDocumentDefault
    wdPart.Close SaveChanges:=False
    Set wdPart = Nothing

    ' ---- 6. Мердж: титул -> body x N -> согласование -> appendix x N ----
    Call AppendPartToMaster(wdApp, wdMaster, titleFile, True)

    For i = 1 To srcCount
        Call AppendPartToMaster(wdApp, wdMaster, bodyFiles(i), False)
    Next i

    Call AppendPartToMaster(wdApp, wdMaster, approvalFile, False)

    For i = 1 To srcCount
        Call AppendPartToMaster(wdApp, wdMaster, appxFiles(i), False)
    Next i

    ' ---- 7. Финальные скаляры + сохранение ----
    Call ApplyConfigScalars(wbMain, wdMaster)

    outPath = BuildOutputPath(wbMain)
    wdMaster.SaveAs2 Filename:=outPath, FileFormat:=wdFormatDocumentDefault

    If optsSaved Then
        On Error Resume Next
        wdApp.Options.PasteFormatBetweenDocuments = optBetween
        wdApp.Options.PasteFormatBetweenStyledDocuments = optStyled
        On Error GoTo Fail
    End If

    On Error Resume Next
    wdApp.ScreenUpdating = True
    On Error GoTo Fail

    If SHOW_RESULT Then
        wdApp.Visible = True
        wdApp.Activate
    Else
        wdMaster.Close SaveChanges:=False
        wdApp.Quit
    End If

    Call CleanupTempDir(tmpDir)

    Application.ScreenUpdating = prevScreen
    Application.DisplayAlerts = prevAlerts

    MsgBox "Готово. Источников обработано: " & srcCount & vbCrLf & _
           "Файл: " & outPath, vbInformation
    Exit Sub

Fail:
    Dim errText As String
    errText = "Ошибка сборки: " & Err.Number & " — " & Err.Description

    On Error Resume Next
    If Not wbData Is Nothing Then wbData.Close SaveChanges:=False
    If Not wdPart Is Nothing Then wdPart.Close SaveChanges:=False
    If Not wdApp Is Nothing Then
        If optsSaved Then
            wdApp.Options.PasteFormatBetweenDocuments = optBetween
            wdApp.Options.PasteFormatBetweenStyledDocuments = optStyled
        End If
        wdApp.ScreenUpdating = True
        wdApp.Visible = True
        wdApp.DisplayAlerts = -1
    End If
    Application.ScreenUpdating = prevScreen
    Application.DisplayAlerts = prevAlerts
    On Error GoTo 0

    MsgBox errText, vbCritical

End Sub


' ============================================================
' МЕРДЖ ОДНОЙ ЧАСТИ В MASTER С СОХРАНЕНИЕМ ОФОРМЛЕНИЯ
' ============================================================
' Посекционный мердж: для каждой секции источника в master создаётся своя
' секция с теми же параметрами страницы, и туда вставляется только её
' содержимое. Это точный аналог ручного «выделил кусок — Ctrl+C — Ctrl+V».
Private Const MERGE_BY_SECTION As Boolean = True

' Вставлять через Selection, а не через Range.
' Range.PasteAndFormat в части случаев игнорирует «сохранить исходное
' форматирование» и применяет стили приёмника. Selection.PasteAndFormat —
' буквально тот же код, что выполняется при ручной вставке мышкой.
Private Const PASTE_VIA_SELECTION As Boolean = True

Private Const wdStory As Long = 6


Private Sub AppendPartToMaster(wdApp As Object, wdMaster As Object, _
                               ByVal filePath As String, ByVal isFirstPart As Boolean)

    Dim srcDoc As Object
    Dim rng As Object
    Dim srcRange As Object
    Dim newSecIdx As Long
    Dim k As Long
    Dim secCount As Long

    If Not MERGE_KEEP_SOURCE_FORMATTING Then
        Call AppendFileToMaster_Legacy(wdMaster, filePath)
        Exit Sub
    End If

    Set srcDoc = wdApp.Documents.Open(Filename:=filePath, ReadOnly:=True, _
                                      AddToRecentFiles:=False, Visible:=False)

    If MERGE_BY_SECTION Then
        secCount = srcDoc.Sections.Count
    Else
        secCount = 1
    End If

    For k = 1 To secCount

        ' --- секция-приёмник ---
        ' Самая первая вставка идёт в уже существующую пустую секцию master,
        ' иначе перед титульником осталась бы пустая страница.
        If isFirstPart And k = 1 Then
            newSecIdx = 1
        Else
            Set rng = wdMaster.Content
            rng.Collapse wdCollapseEnd
            rng.InsertBreak wdSectionBreakNextPage
            newSecIdx = wdMaster.Sections.Count
        End If

        Call UnlinkHeadersFooters(wdMaster.Sections(newSecIdx))
        Call CopyPageSetup(srcDoc.Sections(k).PageSetup, _
                           wdMaster.Sections(newSecIdx).PageSetup)

        If COPY_HEADERS_FOOTERS Then
            Call CopyHeadersFooters(srcDoc.Sections(k), wdMaster.Sections(newSecIdx))
        End If

        ' --- содержимое без замыкающего знака секции / финального ¶ ---
        If MERGE_BY_SECTION Then
            Set srcRange = srcDoc.Sections(k).Range
        Else
            Set srcRange = srcDoc.Content
        End If

        If srcRange.End - 1 > srcRange.Start Then
            Set srcRange = srcDoc.Range(srcRange.Start, srcRange.End - 1)
        End If

        srcRange.Copy

        Call PasteKeepSource(wdApp, wdMaster)

        ' --- PageSetup прогоняем повторно: вставка может принести
        '     свои параметры страницы и сломать альбомную ориентацию ---
        Call CopyPageSetup(srcDoc.Sections(k).PageSetup, _
                           wdMaster.Sections(wdMaster.Sections.Count).PageSetup)

    Next k

    ' Если мерджили целиком — PageSetup последней секции источника хранится
    ' в финальном ¶ и через буфер не переносится.
    If Not MERGE_BY_SECTION Then
        Call CopyPageSetup(srcDoc.Sections(srcDoc.Sections.Count).PageSetup, _
                           wdMaster.Sections(wdMaster.Sections.Count).PageSetup)
        If COPY_HEADERS_FOOTERS Then
            Call UnlinkHeadersFooters(wdMaster.Sections(wdMaster.Sections.Count))
            Call CopyHeadersFooters(srcDoc.Sections(srcDoc.Sections.Count), _
                                    wdMaster.Sections(wdMaster.Sections.Count))
        End If
    End If

    srcDoc.Close SaveChanges:=False
    Set srcDoc = Nothing

End Sub


' Старый способ, оставлен для отката (MERGE_KEEP_SOURCE_FORMATTING = False)
Private Sub AppendFileToMaster_Legacy(wdMaster As Object, ByVal filePath As String)

    Dim rng As Object

    Set rng = wdMaster.Content
    rng.Collapse wdCollapseEnd
    rng.InsertBreak wdSectionBreakNextPage

    Set rng = wdMaster.Content
    rng.Collapse wdCollapseEnd
    rng.InsertFile Filename:=filePath, ConfirmConversions:=False, Link:=False, Attachment:=False

End Sub


' Вставка «сохранить исходное форматирование» с ретраями:
' буфер обмена иногда занят другим процессом.
Private Sub PasteKeepSource(wdApp As Object, wdMaster As Object)

    Dim attempt As Long
    Dim rng As Object

    For attempt = 1 To 5
        Err.Clear
        On Error Resume Next

        If PASTE_VIA_SELECTION Then
            wdMaster.Activate
            wdApp.Selection.EndKey Unit:=wdStory
            wdApp.Selection.Collapse wdCollapseEnd
            wdApp.Selection.PasteAndFormat wdFormatOriginalFormatting
        Else
            Set rng = wdMaster.Content
            rng.Collapse wdCollapseEnd
            rng.PasteAndFormat wdFormatOriginalFormatting
        End If

        If Err.Number = 0 Then
            On Error GoTo 0
            Exit Sub
        End If
        On Error GoTo 0
        DoEvents
        Application.Wait Now + TimeValue("00:00:01")
    Next attempt

    Err.Raise vbObjectError + 513, "AppendPartToMaster", _
              "Не удалось вставить часть документа через буфер обмена."

End Sub


' ============================================================
' ПАРАМЕТРЫ СТРАНИЦЫ И КОЛОНТИТУЛЫ
' ============================================================
Private Sub CopyPageSetup(srcPS As Object, dstPS As Object)

    On Error Resume Next

    dstPS.Orientation = srcPS.Orientation
    dstPS.PaperSize = srcPS.PaperSize
    dstPS.PageWidth = srcPS.PageWidth
    dstPS.PageHeight = srcPS.PageHeight

    dstPS.TopMargin = srcPS.TopMargin
    dstPS.BottomMargin = srcPS.BottomMargin
    dstPS.LeftMargin = srcPS.LeftMargin
    dstPS.RightMargin = srcPS.RightMargin
    dstPS.Gutter = srcPS.Gutter
    dstPS.HeaderDistance = srcPS.HeaderDistance
    dstPS.FooterDistance = srcPS.FooterDistance

    dstPS.MirrorMargins = srcPS.MirrorMargins
    dstPS.VerticalAlignment = srcPS.VerticalAlignment
    dstPS.DifferentFirstPageHeaderFooter = srcPS.DifferentFirstPageHeaderFooter
    dstPS.OddAndEvenPagesHeaderFooter = srcPS.OddAndEvenPagesHeaderFooter

    On Error GoTo 0

End Sub


Private Sub UnlinkHeadersFooters(sec As Object)

    Dim i As Long

    On Error Resume Next
    For i = 1 To 3
        sec.Headers(i).LinkToPrevious = False
        sec.Footers(i).LinkToPrevious = False
    Next i
    On Error GoTo 0

End Sub


Private Sub CopyHeadersFooters(srcSec As Object, dstSec As Object)

    Dim i As Long
    Dim t As String

    On Error Resume Next

    For i = 1 To 3
        t = srcSec.Headers(i).Range.Text
        t = Replace(Replace(Replace(t, Chr(13), ""), Chr(7), ""), Chr(160), " ")
        If Trim(t) <> "" Then
            dstSec.Headers(i).Range.FormattedText = srcSec.Headers(i).Range.FormattedText
        End If

        t = srcSec.Footers(i).Range.Text
        t = Replace(Replace(Replace(t, Chr(13), ""), Chr(7), ""), Chr(160), " ")
        If Trim(t) <> "" Then
            dstSec.Footers(i).Range.FormattedText = srcSec.Footers(i).Range.FormattedText
        End If
    Next i

    On Error GoTo 0

End Sub


' ============================================================
' СБОР DATA-ИСТОЧНИКОВ ИЗ SOURCES
' ============================================================
Private Function CollectDataSources(wbMain As Workbook, _
                                    ByRef srcNums() As String, _
                                    ByRef srcRows() As Long, _
                                    ByRef srcPaths() As String) As Long

    Dim ws As Worksheet
    Dim numCol As Long, fileCol As Long, pathCol As Long
    Dim lastRow As Long
    Dim r As Long
    Dim n As Long

    Dim numTxt As String
    Dim fileTxt As String
    Dim pathTxt As String

    Set ws = wbMain.Worksheets("Sources")

    numCol = FindHeaderColumn(ws, "Номер")
    fileCol = FindHeaderColumn(ws, "Файл")
    pathCol = FindHeaderColumn(ws, "Путь")

    If numCol = 0 Or pathCol = 0 Then
        CollectDataSources = 0
        Exit Function
    End If

    lastRow = ws.Cells(ws.Rows.Count, numCol).End(xlUp).Row

    ReDim srcNums(1 To Application.Max(1, lastRow))
    ReDim srcRows(1 To Application.Max(1, lastRow))
    ReDim srcPaths(1 To Application.Max(1, lastRow))

    n = 0

    For r = 2 To lastRow

        numTxt = Trim(CStr(ws.Cells(r, numCol).Value))

        If numTxt <> "" And InStr(numTxt, ".") = 0 And InStr(numTxt, ",") = 0 Then

            If fileCol > 0 Then
                fileTxt = Trim(CStr(ws.Cells(r, fileCol).Value))
            Else
                fileTxt = "x"
            End If

            pathTxt = Trim(CStr(ws.Cells(r, pathCol).Value))
            pathTxt = Replace(pathTxt, Chr(160), " ")
            pathTxt = Trim(pathTxt)

            If fileTxt <> "" And pathTxt <> "" Then
                If FileExists(pathTxt) Then
                    n = n + 1
                    srcNums(n) = numTxt
                    srcRows(n) = r
                    srcPaths(n) = pathTxt
                End If
            End If
        End If
    Next r

    If n > 0 Then
        ReDim Preserve srcNums(1 To n)
        ReDim Preserve srcRows(1 To n)
        ReDim Preserve srcPaths(1 To n)
    End If

    CollectDataSources = n

End Function


' ============================================================
' СКАЛЯРЫ ИЗ CONFIG
' ============================================================
Public Sub ApplyConfigScalars(wbMain As Workbook, wdDoc As Object)

    Dim protNum As String
    Dim date1 As String
    Dim date1Formatted As String
    Dim date2 As String

    protNum = GetConfigValue(wbMain, "{{ПРОТ_НОМЕР}}")
    date1 = GetConfigValue(wbMain, "{{ДАТА1}}")
    date1Formatted = FormatDateForProtocol(date1)

    If IsDate(date1) Then
        date2 = Format(DateAdd("d", 90, CDate(date1)), "dd.mm.yyyy")
    Else
        date2 = date1
    End If

    Call ReplacePlaceholderInWord(wdDoc, "{{ПРОТ_НОМЕР}}", protNum)
    Call ReplacePlaceholderInWord(wdDoc, "{{ДАТА1ФОРМАТ}}", date1Formatted)
    Call ReplacePlaceholderInWord(wdDoc, "{{ДАТА1}}", date1)
    Call ReplacePlaceholderInWord(wdDoc, "{{ДАТА2}}", date2)

End Sub


' ============================================================
' НОМЕРА РАЗДЕЛОВ: {{I}} и жёстко прописанные 1.1.x
' ============================================================
Public Sub FinalizeSectionNumbers(wdDoc As Object, ByVal sourceNumber As String)

    Dim k As Long
    Dim sn As String

    sn = Trim(sourceNumber)

    Call ReplacePlaceholderInWord(wdDoc, "1.{{I}.", "1." & sn & ".")
    Call ReplacePlaceholderInWord(wdDoc, "{{I}}", sn)

    If FIX_HARDCODED_SECTION_NUMBERS And sn <> "1" Then
        For k = 1 To 4
            Call ReplacePlaceholderInWord(wdDoc, "1.1." & CStr(k), sn_Sect(sn, k))
        Next k
    End If

End Sub

Private Function sn_Sect(ByVal sn As String, ByVal k As Long) As String
    sn_Sect = "1." & sn & "." & CStr(k)
End Function


' ============================================================
' УДАЛЕНИЕ БЛОКА "Приложение № 1.x.4" (рассрочка по ДДУ)
' ============================================================
Private Sub DropDduInstalmentBlock(wdDoc As Object)

    Dim p As Long
    Dim t As String
    Dim startPos As Long
    Dim rngDel As Object

    For p = 1 To wdDoc.Paragraphs.Count
        t = wdDoc.Paragraphs(p).Range.Text
        t = Replace(t, Chr(13), "")
        t = Replace(t, Chr(7), "")
        t = Replace(t, Chr(160), " ")

        If InStr(1, t, "Приложение", vbTextCompare) > 0 And InStr(t, ".4") > 0 Then

            startPos = wdDoc.Paragraphs(p).Range.Start
            If startPos > 0 Then startPos = startPos - 1

            On Error Resume Next
            Set rngDel = wdDoc.Range(startPos, wdDoc.Content.End - 1)
            rngDel.Delete
            On Error GoTo 0

            Exit Sub
        End If
    Next p

End Sub


' ============================================================
' СКОЛЬКО СТРОК ДАННЫХ В VECTOR-БЛОКЕ MAPPER
' ============================================================
Public Function CountVectorRows(wbMain As Workbook, wbData As Workbook, ByVal blockName As String) As Long

    Dim wsMapper As Worksheet
    Dim wsData As Worksheet
    Dim lastMapRow As Long
    Dim r As Long

    Dim sheetName As String
    Dim keyColLetter As String
    Dim startRow As Long
    Dim keyColIdx As Long
    Dim lastDataRow As Long

    Set wsMapper = wbMain.Worksheets("Mapper")
    lastMapRow = wsMapper.Cells(wsMapper.Rows.Count, 2).End(xlUp).Row

    For r = 2 To lastMapRow
        If Trim(CStr(wsMapper.Cells(r, 2).Value)) = blockName Then
            sheetName = Trim(CStr(wsMapper.Cells(r, 5).Value))
            keyColLetter = Trim(CStr(wsMapper.Cells(r, 6).Value))
            If Trim(CStr(wsMapper.Cells(r, 7).Value)) <> "" Then
                startRow = CLng(wsMapper.Cells(r, 7).Value)
            End If
            Exit For
        End If
    Next r

    If sheetName = "" Or keyColLetter = "" Or startRow = 0 Then
        CountVectorRows = 0
        Exit Function
    End If

    On Error GoTo Fail
    Set wsData = wbData.Worksheets(sheetName)
    keyColIdx = ColLetterToIndex(keyColLetter)
    lastDataRow = wsData.Cells(wsData.Rows.Count, keyColIdx).End(xlUp).Row

    If lastDataRow >= startRow Then
        CountVectorRows = lastDataRow - startRow + 1
    Else
        CountVectorRows = 0
    End If
    Exit Function

Fail:
    CountVectorRows = 0

End Function


' ============================================================
' Чистка хвостовых пустых абзацев части
' ============================================================
Private Sub TrimTrailingEmptyParagraphs(wdDoc As Object)

    Dim guard As Long
    Dim t As String

    On Error Resume Next

    For guard = 1 To 20
        If wdDoc.Paragraphs.Count <= 1 Then Exit Sub
        If wdDoc.Paragraphs(wdDoc.Paragraphs.Count).Range.Tables.Count > 0 Then Exit Sub

        t = wdDoc.Paragraphs(wdDoc.Paragraphs.Count).Range.Text
        t = Replace(t, Chr(13), "")
        t = Replace(t, Chr(7), "")
        t = Replace(t, Chr(160), " ")

        If Trim(t) <> "" Then Exit Sub

        wdDoc.Paragraphs(wdDoc.Paragraphs.Count).Range.Delete
    Next guard

End Sub


' ============================================================
' ФАЙЛОВЫЕ УТИЛИТЫ
' ============================================================
Private Function FileExists(ByVal p As String) As Boolean

    Dim res As String

    FileExists = False
    p = Trim(Replace(p, Chr(160), " "))
    If p = "" Then Exit Function
    If Right$(p, 1) = "\" Then Exit Function

    On Error Resume Next
    res = Dir(p)
    On Error GoTo 0

    FileExists = (res <> "")

End Function


Private Function MakeTempDir() As String

    Dim base As String
    Dim p As String

    base = Environ$("TEMP")
    If base = "" Then base = ThisWorkbook.Path
    If Right$(base, 1) <> "\" Then base = base & "\"

    p = base & "protocol_build_" & Format(Now, "yyyymmdd_hhnnss") & "\"

    On Error GoTo Fail
    MkDir Left$(p, Len(p) - 1)
    MakeTempDir = p
    Exit Function

Fail:
    MakeTempDir = ""

End Function


Private Sub CleanupTempDir(ByVal tmpDir As String)

    On Error Resume Next
    If tmpDir = "" Then Exit Sub
    Kill tmpDir & "*.docx"
    RmDir Left$(tmpDir, Len(tmpDir) - 1)

End Sub


' ============================================================
' ДИАГНОСТИКА: где расходятся определения одноимённых стилей
' в двух шаблонах. Запускать вручную, отчёт ложится рядом с main.xlsx.
' ============================================================
Public Sub ReportStyleConflicts()

    Dim wbMain As Workbook
    Dim wdApp As Object
    Dim d1 As Object, d2 As Object
    Dim s1 As Object, s2 As Object
    Dim p1 As String, p2 As String
    Dim outFile As String
    Dim ff As Integer
    Dim nm As String
    Dim cnt As Long

    Set wbMain = ThisWorkbook
    p1 = GetSourcePathByNumber(wbMain, "0.1")                        ' титульник
    p2 = GetSourcePathByNumber(wbMain, MASTER_STYLE_SOURCE_NUMBER)   ' донор стилей

    If Not FileExists(p1) Or Not FileExists(p2) Then
        MsgBox "Не найдены шаблоны для сравнения.", vbExclamation
        Exit Sub
    End If

    Set wdApp = CreateObject("Word.Application")
    wdApp.Visible = False

    Set d1 = wdApp.Documents.Open(Filename:=p1, ReadOnly:=True, AddToRecentFiles:=False, Visible:=False)
    Set d2 = wdApp.Documents.Open(Filename:=p2, ReadOnly:=True, AddToRecentFiles:=False, Visible:=False)

    outFile = wbMain.Path & "\style_conflicts.txt"
    ff = FreeFile
    Open outFile For Output As #ff

    Print #ff, "Стиль | " & d1.Name & " | " & d2.Name

    For Each s1 In d1.Styles
        nm = ""
        On Error Resume Next
        nm = s1.NameLocal
        Set s2 = Nothing
        Set s2 = d2.Styles(nm)
        On Error GoTo 0

        If Not s2 Is Nothing Then
            On Error Resume Next
            If s1.Font.Size <> s2.Font.Size Or s1.Font.Name <> s2.Font.Name Then
                Print #ff, nm & " | " & s1.Font.Name & " " & s1.Font.Size & " | " & _
                                       s2.Font.Name & " " & s2.Font.Size
                cnt = cnt + 1
            End If
            On Error GoTo 0
        End If
    Next s1

    Close #ff

    d1.Close SaveChanges:=False
    d2.Close SaveChanges:=False
    wdApp.Quit

    MsgBox "Расхождений по одноимённым стилям: " & cnt & vbCrLf & outFile, vbInformation

End Sub


Private Function BuildOutputPath(wbMain As Workbook) As String

    Dim protNum As String
    Dim safeNum As String
    Dim i As Long
    Dim bad As Variant

    protNum = GetConfigValue(wbMain, "{{ПРОТ_НОМЕР}}")
    safeNum = protNum

    bad = Array("\", "/", ":", "*", "?", """", "<", ">", "|")
    For i = LBound(bad) To UBound(bad)
        safeNum = Replace(safeNum, CStr(bad(i)), "-")
    Next i

    safeNum = Trim(safeNum)
    If safeNum = "" Then safeNum = "no_num"

    BuildOutputPath = wbMain.Path & "\protocol_" & safeNum & "_" & _
                      Format(Now, "yyyymmdd_hhnn") & ".docx"

End Function
